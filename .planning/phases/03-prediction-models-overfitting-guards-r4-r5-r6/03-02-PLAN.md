```python
"""Model versioning, persistence, and rollback for all prediction models.

Each model (nba_game, nba_totals, nba_props_pts, etc.) is stored independently.
Versioning is integer-based (1, 2, 3, ...). The 'active' version is the one
predict() loads. rollback() reverts to the previous version.

Storage:
    data/models/{model_name}/v{version}.joblib       — scikit-learn model
    data/models/{model_name}/v{version}_cal.joblib   — calibrator (optional)

SQLite model_versions table (Phase 1 schema, D-16) tracks metadata.
"""

import hashlib
import json
import logging
import os
from datetime import datetime, timezone
from pathlib import Path
from typing import Any

import joblib

from src.db.database import Database
from src.models.exceptions import ModelNotFoundError

logger = logging.getLogger(__name__)

BOOTSTRAP_THRESHOLD = 50  # D-08, D-09: minimum OOS predictions for full guard enforcement


class ModelStore:
    """Versioned storage for scikit-learn models with SQLite metadata tracking.

    Args:
        model_name: Unique name (e.g., 'nba_game', 'nba_props_pts').
        store_dir: Directory for .joblib files. Default 'data/models'.
        db_path: SQLite database path. Default 'data/kalshi_bot.db'.
    """

    def __init__(
        self,
        model_name: str,
        store_dir: str = "data/models",
        db_path: str = "data/kalshi_bot.db",
    ) -> None:
        self.model_name = model_name
        self.store_dir = Path(store_dir) / model_name
        self.store_dir.mkdir(parents=True, exist_ok=True)
        self.db = Database(db_path=db_path)

    def save(
        self,
        model: Any,
        calibrator: Any,
        metrics: dict,
        training_data_hash: str,
    ) -> int:
        """Persist model to disk and record metadata in SQLite.

        Sets is_active=0 on all prior versions, then inserts new row with
        is_active=1.  Returns the new version number.

        Args:
            model: Fitted scikit-learn estimator (or CalibratedClassifierCV).
            calibrator: Separate calibrator object, or None if model is already calibrated.
            metrics: Dict with keys like 'test_brier', 'n_train', 'n_test'.
            training_data_hash: MD5 or SHA prefix of training data for reproducibility.

        Returns:
            New version number (1-based integer).
        """
        # Deactivate all previous versions
        self.db.execute(
            "UPDATE model_versions SET is_active = 0 WHERE model_name = ?",
            (self.model_name,),
        )

        # Determine next version
        row = self.db.fetchone(
            "SELECT MAX(version) AS max_version FROM model_versions WHERE model_name = ?",
            (self.model_name,),
        )
        next_version = (row["max_version"] or 0) + 1 if row else 1

        # Save model file
        model_path = self.store_dir / f"v{next_version}.joblib"
        joblib.dump(model, model_path)

        # Save calibrator file if provided
        if calibrator is not None:
            cal_path = self.store_dir / f"v{next_version}_cal.joblib"
            joblib.dump(calibrator, cal_path)

        # Build parameters_json: file paths + feature names if model has them
        parameters = {
            "model_file": str(model_path),
            "calibrator_file": str(self.store_dir / f"v{next_version}_cal.joblib") if calibrator else None,
        }
        if hasattr(model, "feature_names_in_"):
            parameters["feature_names"] = list(model.feature_names_in_)

        self.db.execute(
            """
            INSERT INTO model_versions
                (model_name, version, parameters_json, metrics_json, training_data_hash, is_active, created_at)
            VALUES (?, ?, ?, ?, ?, 1, ?)
            """,
            (
                self.model_name,
                next_version,
                json.dumps(parameters),
                json.dumps(metrics),
                training_data_hash,
                datetime.now(timezone.utc).isoformat(),
            ),
        )

        logger.info(
            "Saved %s v%d (Brier=%.4f, hash=%s)",
            self.model_name, next_version,
            metrics.get("test_brier", float("nan")),
            training_data_hash,
        )
        return next_version

    def load_latest(self) -> tuple:
        """Load the currently active model version from disk.

        Returns:
            (model, calibrator) where calibrator may be None if not saved separately.

        Raises:
            ModelNotFoundError: If no active model exists for this model_name.
        """
        row = self.db.fetchone(
            "SELECT * FROM model_versions WHERE model_name = ? AND is_active = 1 ORDER BY version DESC LIMIT 1",
            (self.model_name,),
        )
        if row is None:
            raise ModelNotFoundError(self.model_name)

        params = json.loads(row["parameters_json"])
        model_path = params["model_file"]
        cal_path = params.get("calibrator_file")

        if not os.path.exists(model_path):
            raise ModelNotFoundError(
                self.model_name,
                version=row["version"],
            )

        model = joblib.load(model_path)
        calibrator = joblib.load(cal_path) if cal_path and os.path.exists(cal_path) else None

        logger.debug("Loaded %s v%d from %s", self.model_name, row["version"], model_path)
        return model, calibrator

    def get_active_version(self) -> int:
        """Return the version number of the currently active model.

        Returns:
            Integer version, or 0 if no active model exists.
        """
        row = self.db.fetchone(
            "SELECT version FROM model_versions WHERE model_name = ? AND is_active = 1",
            (self.model_name,),
        )
        return row["version"] if row else 0

    def rollback(self) -> int:
        """Revert to the previous version by setting it as active.

        Deactivates the current version and activates version = current - 1.

        Returns:
            Version number that is now active.

        Raises:
            ModelNotFoundError: If there is no previous version to roll back to.
        """
        current = self.get_active_version()
        if current <= 1:
            raise ModelNotFoundError(
                self.model_name,
                version=None,
            )
        prev_version = current - 1
        self.db.execute(
            "UPDATE model_versions SET is_active = 0 WHERE model_name = ? AND version = ?",
            (self.model_name, current),
        )
        self.db.execute(
            "UPDATE model_versions SET is_active = 1 WHERE model_name = ? AND version = ?",
            (self.model_name, prev_version),
        )
        logger.info("Rolled back %s from v%d to v%d", self.model_name, current, prev_version)
        return prev_version

    def is_bootstrap_mode(self) -> bool:
        """Return True when fewer than 50 out-of-sample predictions exist (D-08, D-09).

        Counts rows in the predictions table for this model_name. Phase 4 position
        sizer reads this to apply reduced sizing during bootstrap.

        Returns:
            True if OOS prediction count < BOOTSTRAP_THRESHOLD (50).
        """
        row = self.db.fetchone(
            "SELECT COUNT(*) AS n FROM predictions WHERE model_name = ?",
            (self.model_name,),
        )
        n = row["n"] if row else 0
        return n < BOOTSTRAP_THRESHOLD

    def list_versions(self) -> list[dict]:
        """Return all saved versions sorted by version ascending.

        Returns:
            List of dicts with keys: version, is_active, created_at, metrics_json.
        """
        return self.db.fetchall(
            "SELECT version, is_active, created_at, metrics_json FROM model_versions WHERE model_name = ? ORDER BY version ASC",
            (self.model_name,),
        )
```

Temperature bracket model per R5.1, R5.2, R5.4, R5.5, R5.6, R5.7, D-10, D-11, D-12.

Kalshi temperature 6-bracket market structure (example for NYC April):
```python
# Brackets are defined per market — the model takes bracket ranges as input.
# Example NYC high temp brackets: [<45, 45-49, 50-54, 55-59, 60-64, >=65]
# Stored as list of (low, high) tuples where None means no bound.
```

**Key design decisions to implement:**
- D-10: `bias_offsets[station][month] = mean(tmax_f - nws_forecast_high)` from NOAA vs NWS comparison
- D-11: After bias correction: `adjusted_mu = nws_point_forecast + bias_offset`. Then `sigma = std(tmax_f - nws_forecast_high)` per station. `P(bracket_i) = scipy.stats.norm.cdf(hi, mu, sigma) - scipy.stats.norm.cdf(lo, mu, sigma)`
- D-12: 12 months (not 4 seasons) — `bias_offsets["KNYC"][4]` for April (month integer 1-12)
- R5.7 DST: Kalshi uses local standard time (LST), not midnight-to-midnight. For NYC, the "high temperature of the day" is measured 6am-6pm LST. Use `zoneinfo.ZoneInfo(timezone)` + `timedelta(hours=6)` offset when extracting forecast high.

```python
import logging
from datetime import date, datetime, timedelta
from typing import Optional
import numpy as np
from scipy import stats
from src.data.weather.station_map import STATIONS
from src.db.database import Database
from src.models.exceptions import ModelNotFoundError, InsufficientDataError

logger = logging.getLogger(__name__)


class WeatherTempModel:
    """Temperature bracket probability model using NWS ensemble + bias correction.

    Workflow:
        1. fit_bias_correction() — learn additive offsets from NOAA historical data
        2. predict_brackets(station, forecast_date, brackets) — return per-bracket probs
        3. predict_threshold(station, threshold_f, forecast_date) — return P(high > X)

    All predictions use Gaussian fit centered on bias-corrected NWS point forecast (D-11).
    Bias correction: per station × month (D-10, D-12). 12-month granularity.
    DST handling: NWS hourly forecast filtered to local standard time window (R5.7).
    """

    def __init__(self, db_path: str = "data/kalshi_bot.db") -> None:
        self.db = Database(db_path=db_path)
        # bias_offsets[station_code][month_int] = (mean_offset_f, std_error_f)
        # Populated by fit_bias_correction(). None until fit is called.
        self.bias_offsets: dict = {}
        self._is_fitted = False

    def fit_bias_correction(self, min_observations: int = 30) -> dict:
        """Learn additive bias corrections from NOAA historical vs NWS forecasts.

        Computes per station × month offsets:
            bias_offset[station][month] = mean(actual_tmax - nws_forecast_high)

        Since NWS forecast history is not stored, this uses NOAA tmax_f directly
        as the 'actual' and computes within-month variance as sigma.

        In a fully trained system, this method would be called after accumulating
        historical NWS forecasts alongside NOAA actuals. For Phase 3, we use the
        variance of NOAA tmax within each station × month as the sigma (forecast
        uncertainty proxy) and set the mean offset to 0.0 as the initial prior
        (will be non-zero once NWS forecast history accumulates).

        Args:
            min_observations: Minimum NOAA observations per station × month.

        Returns:
            Dict mapping station_code -> {month: (offset_f, sigma_f)}.

        Raises:
            InsufficientDataError: If any station has fewer than min_observations.
        """
        import statistics

        result = {}
        for station_code in STATIONS:
            rows = self.db.fetchall(
                "SELECT date, tmax_f FROM noaa_daily_weather WHERE station_code = ? AND tmax_f IS NOT NULL ORDER BY date ASC",
                (station_code,),
            )
            if len(rows) < min_observations:
                logger.warning(
                    "Station %s has only %d NOAA records (need %d). Using default sigma=5.0°F.",
                    station_code, len(rows), min_observations,
                )
                result[station_code] = {m: (0.0, 5.0) for m in range(1, 13)}
                continue

            # Group by month
            by_month: dict[int, list[float]] = {m: [] for m in range(1, 13)}
            for row in rows:
                month = int(row["date"][5:7])
                by_month[month].append(row["tmax_f"])

            station_offsets = {}
            for month, temps in by_month.items():
                if len(temps) < 5:
                    station_offsets[month] = (0.0, 5.0)
                else:
                    # Mean offset = 0.0 (no historical NWS forecasts stored yet)
                    # Sigma = std dev of actual tmax values as uncertainty proxy
                    sigma = statistics.stdev(temps) if len(temps) > 1 else 5.0
                    station_offsets[month] = (0.0, sigma)

            result[station_code] = station_offsets
            logger.info("Fitted bias correction for %s: %d months, avg sigma=%.1f°F",
                       station_code, len(station_offsets),
                       sum(v[1] for v in station_offsets.values()) / len(station_offsets))

        self.bias_offsets = result
        self._is_fitted = True
        return result

    def _get_nws_point_forecast(self, station: str, forecast_date: str) -> float:
        """Extract point forecast high temperature for the given date from NWS.

        Args:
            station: Station code (e.g., 'KNYC').
            forecast_date: ISO date string 'YYYY-MM-DD'.

        Returns:
            Point forecast high temperature in °F for forecast_date (LST window, R5.7).
        """
        from src.data.weather.nws import get_nws_hourly

        station_info = STATIONS[station]
        hourly = get_nws_hourly(station_info["lat"], station_info["lon"])

        # R5.7: Use local standard time 6am-6pm window for daily high
        # Convert UTC forecast times to local standard time (no DST adjustment)
        # Local standard time = UTC + offset (fixed, not daylight saving)
        # For simplicity, use UTC window approximation:
        # NYC (EST = UTC-5): 6am LST = 11:00 UTC; 6pm LST = 23:00 UTC
        # CHI (CST = UTC-6): 6am LST = 12:00 UTC; 6pm LST = 00:00 UTC+1
        # MIA (EST = UTC-5): same as NYC
        # AUS (CST = UTC-6): same as CHI
        tz_offsets = {"America/New_York": -5, "America/Chicago": -6}
        offset_hours = tz_offsets.get(station_info.get("timezone", "America/New_York"), -5)
        # Window start: 6am LST = (6 - offset_hours) UTC
        window_start_utc = 6 - offset_hours  # e.g., for EST: 6-(-5) = 11 UTC
        window_end_utc = 18 - offset_hours    # e.g., for EST: 18-(-5) = 23 UTC

        target_date = forecast_date  # 'YYYY-MM-DD'
        temps_in_window = []
        for period in hourly:
            start_time = period.get("startTime", "")
            if not start_time.startswith(target_date):
                continue
            try:
                hour_utc = int(start_time[11:13])
            except (ValueError, IndexError):
                continue
            if window_start_utc <= hour_utc < window_end_utc:
                temp = period.get("temperature")
                if temp is not None:
                    temps_in_window.append(float(temp))

        if not temps_in_window:
            # Fall back to daytime period forecast
            for period in hourly:
                if period.get("startTime", "").startswith(target_date):
                    t = period.get("temperature")
                    if t is not None:
                        return float(t)
            return 55.0  # Last resort default

        return max(temps_in_window)

    def predict_brackets(
        self,
        station: str,
        forecast_date: str,
        brackets: Optional[list[tuple]] = None,
    ) -> dict[str, float]:
        """Return probability distribution over temperature brackets.

        Args:
            station: Station code (e.g., 'KNYC').
            forecast_date: ISO date string 'YYYY-MM-DD'.
            brackets: List of (low, high) tuples defining bracket ranges.
                      None means open-ended. Default Kalshi 6-bracket structure.
                      Example: [(None, 45), (45, 50), (50, 55), (55, 60), (60, 65), (65, None)]

        Returns:
            Dict mapping bracket label (e.g., '<45', '45-49', '>=65') to probability float.
            Values sum to approximately 1.0.

        Raises:
            ModelNotFoundError: If fit_bias_correction() was never called.
        """

                prob = float(stats.norm.cdf(hi, loc=mu, scale=sigma))
            elif lo is not None and hi is None:
                label = f">={lo}"
                prob = float(1.0 - stats.norm.cdf(lo, loc=mu, scale=sigma))
            elif lo is not None and hi is not None:
                label = f"{lo}-{hi-1}"
                prob = float(stats.norm.cdf(hi, loc=mu, scale=sigma) - stats.norm.cdf(lo, loc=mu, scale=sigma))
            else:
                label = "all"
                prob = 1.0
            result[label] = max(0.0, min(1.0, prob))

        # Normalize to exactly sum to 1.0
        total = sum(result.values())
        if total > 0:
            result = {k: v / total for k, v in result.items()}

        return result

        self,

        threshold_f: float,
        forecast_date: str,
    ) -> float:

            forecast_date: ISO date string 'YYYY-MM-DD'.
```

```python
"""Precipitation threshold probability model.

Combines NWS probabilistic precipitation with NOAA historical base rates.
predict() returns P(precip > threshold_in) for a given station and date.
"""

import logging
from typing import Optional
import numpy as np
from src.data.weather.station_map import STATIONS
from src.db.database import Database

logger = logging.getLogger(__name__)


class WeatherPrecipModel:
    """Precipitation threshold model: P(precip > threshold_in).

    Combines NWS probability of precipitation (PoP) with historical climatological
    base rates from NOAA. Lead-time weighting per R5.5: closer forecasts weighted
    more heavily than distant ones.

    For Phase 3, uses NWS PoP as primary signal with historical calibration applied
    as a Bayesian update: adjusted_prob = alpha * nws_pop + (1-alpha) * historical_rate
    where alpha = max(0.3, 1.0 - days_until / 7.0) (lead-time weight, R5.5).
    """

    def __init__(self, db_path: str = "data/kalshi_bot.db") -> None:
        self.db = Database(db_path=db_path)
        # historical_rates[station][month] = historical P(precip > 0.1in)
        # Populated lazily from NOAA data
        self._historical_rates: dict = {}

    def _get_historical_rate(self, station: str, month: int, threshold_in: float) -> float:
        """Compute historical P(precip > threshold_in) for station × month from NOAA."""
        cache_key = (station, month, threshold_in)
        if cache_key not in self._historical_rates:
            rows = self.db.fetchall(
                """SELECT prcp_in FROM noaa_daily_weather
                   WHERE station_code = ? AND CAST(strftime('%m', date) AS INTEGER) = ?
                   AND prcp_in IS NOT NULL""",
                (station, month),
            )
            if not rows:
                self._historical_rates[cache_key] = 0.3  # default base rate
            else:
                precip_vals = [r["prcp_in"] for r in rows]
                rate = sum(1 for v in precip_vals if v > threshold_in) / len(precip_vals)
                self._historical_rates[cache_key] = rate
        return self._historical_rates[cache_key]

    def _get_nws_pop(self, station: str, forecast_date: str) -> float:
        """Get NWS probability of precipitation for forecast_date from hourly data."""
        from src.data.weather.nws import get_nws_forecast
        station_info = STATIONS[station]
        try:
            periods = get_nws_forecast(station_info["lat"], station_info["lon"])
        except Exception as exc:
            logger.warning("NWS forecast fetch failed for %s: %s", station, exc)
            return 0.3  # fallback

        for period in periods:
            start = period.get("startTime", "")
            if forecast_date in start:
                detail = period.get("detailedForecast", "")
                # NWS includes "Chance of precipitation is X%" in detailed forecast
                import re
                match = re.search(r"chance of precipitation is (\d+)%", detail, re.IGNORECASE)
                if match:
                    return float(match.group(1)) / 100.0
                # Also check probabilityOfPrecipitation if present
                pop = period.get("probabilityOfPrecipitation", {})
                if isinstance(pop, dict) and pop.get("value") is not None:
                    return float(pop["value"]) / 100.0
        return 0.3  # fallback if no matching period

    def predict(
        self,
        station: str,
        threshold_in: float,
        forecast_date: str,
        days_until: int = 1,
    ) -> float:
        """Return P(daily_precip > threshold_in) for a given station and date.

        Args:
            station: Station code (e.g., 'KNYC').
            threshold_in: Precipitation threshold in inches (e.g., 0.1 for trace).
            forecast_date: ISO date string 'YYYY-MM-DD'.
            days_until: Days until the forecast date (for lead-time weighting, R5.5).

        Returns:
            Float in [0.0, 1.0].
        """
        month = int(forecast_date[5:7])
        historical_rate = self._get_historical_rate(station, month, threshold_in)
        nws_pop = self._get_nws_pop(station, forecast_date)

        # Lead-time weight: closer forecasts trust NWS more (R5.5)
        alpha = max(0.3, 1.0 - days_until / 7.0)
        combined = alpha * nws_pop + (1.0 - alpha) * historical_rate

        return max(0.0, min(1.0, combined))
```
