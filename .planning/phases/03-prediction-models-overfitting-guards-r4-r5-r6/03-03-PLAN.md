```python
"""Temporal train/test/holdout splitter with write-protection.

Splits data chronologically (no shuffling) in 60/20/20 ratio by default.
Holdout write-protection: access_holdout() raises PermissionError unless
explicitly allowed — guards against accidental holdout contamination (R6.1).

Anti-patterns prevented:
- No random shuffling (would break causality in time series)
- No future leakage (chronological order preserved)
- No holdout access during training loops
"""

import logging
from typing import Optional
import numpy as np
import pandas as pd

logger = logging.getLogger(__name__)


class TemporalSplitter:
    """Chronological 60/20/20 train/test/holdout splitter.

    Args:
        dates: Array-like of datetime values, sorted ascending.
        test_size: Fraction for test set. Default 0.2.
        holdout_size: Fraction for holdout set. Default 0.2.
    """

    def __init__(
        self,
        dates,
        test_size: float = 0.2,
        holdout_size: float = 0.2,
    ) -> None:
        dates_arr = np.asarray(dates)
        n = len(dates_arr)
        if n < 10:
            raise ValueError(f"Need at least 10 observations for temporal split, got {n}")

        self._n = n
        self._train_end = int(n * (1.0 - test_size - holdout_size))
        self._test_end = self._train_end + int(n * test_size)
        # Remainder goes to holdout (avoids off-by-one rounding losses)
        self._holdout_end = n

        logger.debug(
            "TemporalSplitter: n=%d, train=%d, test=%d, holdout=%d",
            n, self._train_end,
            self._test_end - self._train_end,
            self._holdout_end - self._test_end,
        )

    def split(self, X: np.ndarray, y: np.ndarray) -> tuple:
        """Return chronological (X_train, X_test, X_holdout), (y_train, y_test, y_holdout).

        Args:
            X: Feature matrix, shape (n_samples, n_features).
            y: Labels array, shape (n_samples,).

        Returns:
            Tuple of 2 triples: (X splits), (y splits). All arrays are numpy arrays.
        """
        X = np.asarray(X)
        y = np.asarray(y)
        return (
            (X[:self._train_end], X[self._train_end:self._test_end], X[self._test_end:]),
            (y[:self._train_end], y[self._train_end:self._test_end], y[self._test_end:]),
        )

    def access_holdout(self, allow_holdout: bool = False) -> tuple:
        """Return holdout indices — raises PermissionError unless explicitly allowed.

        The holdout set must only be accessed for final pre-deployment validation.
        Calling this during a training loop is a bug.

        Args:
            allow_holdout: Must be True to access holdout data.

        Returns:
            (holdout_start_idx, holdout_end_idx) tuple.

        Raises:
            PermissionError: If allow_holdout is False.
        """
        if not allow_holdout:
            raise PermissionError(
                "Holdout data is write-protected. "
                "Call access_holdout(allow_holdout=True) only for final deployment validation. "
                "Never access holdout during training or hyperparameter tuning."
            )
        logger.warning("Holdout accessed! Ensure this is final deployment validation only.")
        return (self._test_end, self._holdout_end)
```

```python
"""Walk-forward rolling window validator.

Implements train-on-W, test-on-W+1, slide-forward pattern for realistic
out-of-sample evaluation. Uses sklearn.model_selection.TimeSeriesSplit
to ensure no future leakage (R6.2).
"""

import logging
from typing import Callable, Any
import numpy as np
from sklearn.model_selection import TimeSeriesSplit

logger = logging.getLogger(__name__)


class WalkForwardValidator:
    """Rolling window validation: train on W, test on W+1, slide forward.

    Args:
        n_splits: Number of non-overlapping windows. Default 4.
    """

    def __init__(self, n_splits: int = 4) -> None:
        if n_splits < 2:
            raise ValueError(f"n_splits must be >= 2, got {n_splits}")
        self.n_splits = n_splits
        self._splitter = TimeSeriesSplit(n_splits=n_splits)

    def validate(
        self,
        X: np.ndarray,
        y: np.ndarray,
        train_fn: Callable,
        test_fn: Callable,
    ):
        """Yield per-window metrics for each walk-forward window.

        Args:
            X: Feature matrix, shape (n_samples, n_features).
            y: Labels array, shape (n_samples,).
            train_fn(X_train, y_train) -> model: Returns trained model.
            test_fn(model, X_test, y_test) -> dict: Returns metrics dict.

        Yields:
            Dict with keys:
                - window: Window number (1-based)
                - train_size: Number of training samples
                - test_size: Number of test samples
                - train_end_idx: Last training index (for temporal assertions)
                - test_start_idx: First test index
                - metrics: Dict returned by test_fn (must include 'brier_score')
        """
        X = np.asarray(X)
        y = np.asarray(y)

        for window_num, (train_idx, test_idx) in enumerate(self._splitter.split(X), start=1):
            X_train, X_test = X[train_idx], X[test_idx]
            y_train, y_test = y[train_idx], y[test_idx]

            # Strict temporal ordering assertion
            assert train_idx[-1] < test_idx[0], (
                f"Window {window_num}: last train index ({train_idx[-1]}) must be "
                f"< first test index ({test_idx[0]}). TimeSeriesSplit bug?"
            )

            logger.debug(
                "Walk-forward window %d: train=%d samples, test=%d samples",
                window_num, len(train_idx), len(test_idx),
            )

            model = train_fn(X_train, y_train)
            metrics = test_fn(model, X_test, y_test)

            yield {
                "window": window_num,
                "train_size": len(train_idx),
                "test_size": len(test_idx),
                "train_end_idx": int(train_idx[-1]),
                "test_start_idx": int(test_idx[0]),
                "metrics": metrics,
            }
```

```python
"""Validation metrics for prediction model evaluation.

Provides Brier score, accuracy, and calibration error functions used by
walk-forward validator and the analytics layer (Phase 5).
"""

import numpy as np


def brier_score(probs: np.ndarray, labels: np.ndarray) -> float:
    """Compute Brier score: mean squared error between predicted probs and labels.

    Lower is better. Perfect model = 0.0. Random (0.5 always) = 0.25.
    Calibrated probabilities will score lower than raw model output.

    Args:
        probs: Predicted probabilities array, shape (n,), values in [0.0, 1.0].
        labels: Binary labels array, shape (n,), values 0 or 1.

    Returns:
        Float in [0.0, 1.0]. Typical good model: 0.15-0.22.
    """
    probs = np.asarray(probs, dtype=float)
    labels = np.asarray(labels, dtype=float)
    if len(probs) == 0:
        raise ValueError("Cannot compute Brier score on empty arrays")
    return float(np.mean((probs - labels) ** 2))


def accuracy(probs: np.ndarray, labels: np.ndarray, threshold: float = 0.5) -> float:
    """Compute accuracy: fraction of correct classifications.

    Args:
        probs: Predicted probabilities array.
        labels: Binary labels array.
        threshold: Classification threshold. Default 0.5.

    Returns:
        Float in [0.0, 1.0].
    """
    probs = np.asarray(probs, dtype=float)
    labels = np.asarray(labels, dtype=float)
    predictions = (probs >= threshold).astype(float)
    return float(np.mean(predictions == labels))


def calibration_error(
    probs: np.ndarray,
    labels: np.ndarray,
    n_bins: int = 10,
) -> float:
    """Compute Expected Calibration Error (ECE).

    ECE measures whether predicted probabilities match empirical frequencies.
    Lower is better. Well-calibrated model < 0.05.

    Args:
        probs: Predicted probabilities, shape (n,).
        labels: Binary labels, shape (n,).
        n_bins: Number of bins for reliability diagram. Default 10.

    Returns:
        Float ECE value. Typical range 0.01-0.15.
    """
    probs = np.asarray(probs, dtype=float)
    labels = np.asarray(labels, dtype=float)
    bins = np.linspace(0.0, 1.0, n_bins + 1)
    ece = 0.0
    n = len(probs)
    for i in range(n_bins):
        mask = (probs >= bins[i]) & (probs < bins[i + 1])
        if mask.sum() == 0:
            continue
        bin_prob = probs[mask].mean()
        bin_acc = labels[mask].mean()
        bin_weight = mask.sum() / n
        ece += bin_weight * abs(bin_prob - bin_acc)
    return float(ece)
```

```python
"""Overfitting protection guards (R6.1-R6.8).

All guards raise GuardViolation when a deployment criterion is violated.
During bootstrap mode (< 50 OOS predictions), guards log status but do not
block trading — only model deployment is blocked (D-08).

Parameter values: exactly as specified in REQUIREMENTS.md §R6 (D-09).
Force override: bypasses the guard and logs reason to improvements table.
"""

import logging
from datetime import datetime, timezone, timedelta
from typing import Optional
import numpy as np
from scipy import stats

from src.models.exceptions import GuardViolation

logger = logging.getLogger(__name__)

# R6 parameter constants (D-09: exact values, do not soften)
MIN_OOS_PREDICTIONS = 50          # R6.3
MAX_PARAMETER_CHANGE_PCT = 0.20   # R6.4 — 20%
COOLDOWN_TRADES = 30              # R6.5
STALENESS_DAYS = 30               # R6.6
SIGNIFICANCE_THRESHOLD = 0.05     # R6.7


class OverfittingGuards:
    """Enforcement layer for all overfitting protection rules.

    Args:
        db: Database instance for cooldown check. None = no cooldown check possible.
    """

    def __init__(self, db=None) -> None:
        self.db = db

    def check_sample_size(self, n_oos: int) -> None:
        """Raise GuardViolation if n_oos < 50 (R6.3, D-09).

        Args:
            n_oos: Number of out-of-sample predictions evaluated.

        Raises:
            GuardViolation: If n_oos < MIN_OOS_PREDICTIONS.
        """
        if n_oos < MIN_OOS_PREDICTIONS:
            raise GuardViolation(
                guard_name="sample_size_gate",
                message=(
                    f"Insufficient out-of-sample predictions: {n_oos} < {MIN_OOS_PREDICTIONS} required. "
                    f"Trade in bootstrap mode (reduced sizing) until {MIN_OOS_PREDICTIONS} OOS predictions accumulate."
                ),
                details={"n_oos": n_oos, "required": MIN_OOS_PREDICTIONS},
            )
        logger.debug("Sample size gate passed: %d >= %d OOS predictions", n_oos, MIN_OOS_PREDICTIONS)

    def check_dampening(self, current_value: float, proposed_value: float) -> None:
        """Raise GuardViolation if parameter change > 20% (R6.4, D-09).

        Args:
            current_value: Current parameter value (non-zero).
            proposed_value: Proposed new parameter value.

        Raises:
            GuardViolation: If abs change > MAX_PARAMETER_CHANGE_PCT.
            ValueError: If current_value is zero (change is undefined).
        """
        if current_value == 0:
            raise ValueError("current_value cannot be zero (change percentage is undefined)")
        pct_change = abs(proposed_value - current_value) / abs(current_value)
        if pct_change > MAX_PARAMETER_CHANGE_PCT:
            raise GuardViolation(
                guard_name="dampening",
                message=(
                    f"Parameter change of {pct_change:.1%} exceeds maximum {MAX_PARAMETER_CHANGE_PCT:.0%}. "
                    f"Current: {current_value}, Proposed: {proposed_value}. "
                    f"Apply change in smaller increments."
                ),
                details={
                    "current_value": current_value,
                    "proposed_value": proposed_value,
                    "pct_change": pct_change,
                    "max_allowed": MAX_PARAMETER_CHANGE_PCT,
                },
            )
        logger.debug("Dampening check passed: %.1f%% change within limit", pct_change * 100)

    def check_cooldown(self) -> None:
        """Raise GuardViolation if last applied improvement has < 30 post_apply_trades (R6.5).

        Queries the improvements table for the most recently 'applied' improvement.
        If its post_apply_trades < COOLDOWN_TRADES, raises GuardViolation.

        Raises:
            GuardViolation: If in cooldown window.
            RuntimeError: If db is None (cannot check cooldown without database).
        """
        if self.db is None:
            logger.warning("No database provided to OverfittingGuards — skipping cooldown check")
            return

        row = self.db.fetchone(
            """SELECT id, post_apply_trades, applied_at FROM improvements
               WHERE status = 'applied'
               ORDER BY applied_at DESC LIMIT 1""",
        )
        if row is None:
            logger.debug("Cooldown check: no applied improvements found, skipping.")
            return

        post_apply = row.get("post_apply_trades", 0) or 0
        if post_apply < COOLDOWN_TRADES:
            raise GuardViolation(
                guard_name="cooldown",
                message=(
                    f"Improvement cooldown active: only {post_apply} settled trades since last change "
                    f"(need {COOLDOWN_TRADES}). Wait for more data before next deployment."
                ),
                details={
                    "post_apply_trades": post_apply,
                    "required": COOLDOWN_TRADES,
                    "last_improvement_id": row.get("id"),
                },
            )
        logger.debug("Cooldown check passed: %d post-apply trades >= %d required", post_apply, COOLDOWN_TRADES)

    def check_staleness(self, last_validated_at: str) -> None:
        """Raise GuardViolation if parameters not re-validated in 30+ days (R6.6, D-09).

        Args:
            last_validated_at: ISO 8601 UTC timestamp of last validation run.

        Raises:
            GuardViolation: If days since validation >= STALENESS_DAYS.
        """
        try:
            last_dt = datetime.fromisoformat(last_validated_at.replace("Z", "+00:00"))
        except (ValueError, AttributeError) as exc:
            raise ValueError(f"Invalid last_validated_at format: {last_validated_at!r}") from exc

        now = datetime.now(timezone.utc)
        days_stale = (now - last_dt).days
        if days_stale >= STALENESS_DAYS:
            raise GuardViolation(
                guard_name="staleness",
                message=(
                    f"Model parameters are stale: {days_stale} days since last validation "
                    f"(limit: {STALENESS_DAYS} days). Retrain or re-validate before deployment."
                ),
                details={
                    "days_stale": days_stale,
                    "last_validated_at": last_validated_at,
                    "limit_days": STALENESS_DAYS,
                },
            )
        logger.debug("Staleness check passed: %d days since last validation", days_stale)

    def check_significance(self, p_value: float) -> None:
        """Raise GuardViolation if p_value >= 0.05 (R6.7, D-09).

        Used after binomial test for win rate or t-test for edge.

        Args:
            p_value: p-value from statistical significance test.

        Raises:
            GuardViolation: If p_value >= SIGNIFICANCE_THRESHOLD.
        """
        if p_value >= SIGNIFICANCE_THRESHOLD:
            raise GuardViolation(
                guard_name="significance",
                message=(
                    f"Result not statistically significant: p={p_value:.4f} >= {SIGNIFICANCE_THRESHOLD} threshold. "
                    f"Cannot distinguish model edge from chance at 95% confidence."
                ),
                details={"p_value": p_value, "threshold": SIGNIFICANCE_THRESHOLD},
            )
        logger.debug("Significance check passed: p=%.4f < %.2f", p_value, SIGNIFICANCE_THRESHOLD)

    def check_regime(self, sample1: np.ndarray, sample2: np.ndarray) -> None:
        """Alert on distributional shift via Kolmogorov-Smirnov test (R6.8).

        Compares two samples of model inputs (e.g., recent vs historical feature
        distributions). If KS test detects significant shift, raises GuardViolation
        to alert rather than silently fit to new regime.

        Args:
            sample1: Historical feature sample (1D array).
            sample2: Recent feature sample (1D array).

        Raises:
            GuardViolation: If KS test p-value < 0.05 (distributional shift detected).
        """
        sample1 = np.asarray(sample1, dtype=float)
        sample2 = np.asarray(sample2, dtype=float)
        ks_stat, p_value = stats.ks_2samp(sample1, sample2)
        if p_value < SIGNIFICANCE_THRESHOLD:
            raise GuardViolation(
                guard_name="regime_change",
                message=(
                    f"Regime change detected: KS test p={p_value:.4f} < {SIGNIFICANCE_THRESHOLD}. "
                    f"KS statistic={ks_stat:.4f}. Investigate distributional shift before retraining."
                ),
                details={"ks_stat": ks_stat, "p_value": p_value, "sample1_size": len(sample1), "sample2_size": len(sample2)},
            )
        logger.debug("Regime check passed: KS p=%.4f (no significant shift)", p_value)

    def force_override(
        self,

        reason: str,
        model_name: str,

        """
        if not reason or not reason.strip():

        if self.db is None:
            raise RuntimeError("Cannot log force override without database connection")

        import json
```

```python
def run_train(model_name: str, force_reason: Optional[str] = None) -> None:
    """Train one or all models, running overfitting guards before deployment.
```
